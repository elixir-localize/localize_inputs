    schema "products" do
  field :quantity, :integer
  field :rating,   :decimal
end

def changeset(product, attrs) do
  product
  |> Ecto.Changeset.cast(attrs, [:quantity, :rating])
  |> Localize.Inputs.Changeset.validate_number(:quantity, min: 1)
  |> Localize.Inputs.Changeset.validate_number(:rating, min: 0, max: 5, decimals: 1)
end

    validate_number(changeset, field, options \\ [])

    @spec validate_number(Ecto.Changeset.t(), atom(), Keyword.t()) :: Ecto.Changeset.t()

    Validates a Decimal / integer field with the rules from Localize.Inputs.Validator.validate_number/2.

    Arguments

    • changeset is an Ecto.Changeset.t/0.

    • field is the field name as an atom.

    • options is a keyword list of options forwarded to Localize.Inputs.Validator.validate_number/2.

    Options

    • :required — when true, nil is rejected.

    • :min — minimum allowed value.

    • :max — maximum allowed value.

    • :decimals — maximum number of fractional digits.

    Returns

    • The changeset, with one error added per failing check.

    Examples

    iex> changeset = Ecto.Changeset.cast({%{}, %{quantity: :integer}}, %{"quantity" => 5}, [:quantity])
iex> changeset = Localize.Inputs.Changeset.validate_number(changeset, :quantity, min: 1, max: 10)
iex> changeset.valid?
true
